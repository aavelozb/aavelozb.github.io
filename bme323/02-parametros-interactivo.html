Parámetros estáticos y dinámicos

mulberry32 = (a) => () => {
  a |= 0; a = a + 0x6D2B79F5 | 0;
  let t = Math.imul(a ^ a >>> 15, 1 | a);
  t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
  return ((t ^ t >>> 14) >>> 0) / 4294967296;
}

muestrasBase = {
  const rand = mulberry32(2024);
  const arr = [];
  for (let i = 0; i < 60; i++) {
    const u1 = Math.max(rand(), 1e-9), u2 = rand();
    arr.push({
      z: Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2),
      j: rand()
    });
  }
  return arr;
}

Parámetros estáticos

Exactitud, precisión y resolución

Un sensor de presión mide repetidamente una presión verdadera de 120 mmHg.

  • El sesgo desplaza todas las lecturas: empeora la exactitud.
  • La dispersión (σ) ensancha la nube de lecturas: empeora la precisión (reproducibilidad).
  • La resolución es el incremento más pequeño que el instrumento puede distinguir: las lecturas se “cuantizan” a ese paso.

Note que un instrumento puede ser muy preciso (lecturas casi idénticas) y a la vez inexacto (todas desplazadas del valor verdadero).

viewof sesgo = Inputs.range([-15, 15], { step: 0.5, value: 0, label: "Sesgo (mmHg)" })
viewof sigma = Inputs.range([0.2, 10], { step: 0.2, value: 3, label: "Dispersión σ (mmHg)" })
viewof resol = Inputs.radio([0.1, 0.5, 1, 2, 5], { value: 0.5, label: "Resolución (mmHg)" })
lecturas = muestrasBase.map(d => ({
  valor: Math.round((120 + sesgo + sigma * d.z) / resol) * resol,
  y: d.j
}))

mediaLect = lecturas.reduce((s, d) => s + d.valor, 0) / lecturas.length

sdLect = Math.sqrt(
  lecturas.reduce((s, d) => s + (d.valor - mediaLect) ** 2, 0) / (lecturas.length - 1)
)

})

Curva de calibración: sensibilidad, derivas, linealidad e histéresis

La sensibilidad estática es la pendiente de la curva de calibración entrada–salida. Sobre ella pueden actuar señales modificantes que producen deriva de cero (la curva se traslada) y deriva de sensibilidad (la pendiente cambia). La no linealidad curva la relación entrada–salida, y la histéresis hace que la salida sea distinta cuando la entrada sube y cuando baja.

viewof sens = Inputs.range([0.5, 2], { step: 0.05, value: 1, label: "Sensibilidad K (mV/mmHg)" })
viewof derivaCero = Inputs.range([-20, 20], { step: 1, value: 0, label: "Deriva de cero (mV)" })
viewof derivaSens = Inputs.range([-0.5, 0.5], { step: 0.05, value: 0, label: "Deriva de sensibilidad" })
viewof noLin = Inputs.range([0, 1], { step: 0.05, value: 0, label: "No linealidad" })
viewof hist = Inputs.range([0, 1], { step: 0.05, value: 0, label: "Histéresis" })
curvaCal = {
  const pts = [];
  const n = 80;
  for (let i = 0; i <= n; i++) {
    const x = 100 * i / n;
    const base = (sens + derivaSens) * x + derivaCero + noLin * 0.004 * x * (x - 100) * (sens);
    const lazo = hist * 0.012 * x * (100 - x) * sens;
    pts.push({ x, y: sens * x, serie: "curva nominal (K·x)" });
    pts.push({ x, y: base + lazo, serie: "entrada subiendo ↑" });
    pts.push({ x, y: base - lazo, serie: "entrada bajando ↓" });
  }
  return pts;
}

Plot.plot({
  style: estiloPlot,
  height: 340,
  grid: true,
  x: { label: "Entrada x — presión (mmHg)" },
  y: { label: "Salida y (mV)" },
  color: {
    legend: true,
    domain: ["curva nominal (K·x)", "entrada subiendo ↑", "entrada bajando ↓"],
    range: ["#7f7f7f", "#d62728", "#4c78a8"]
  },
  marks: [
    Plot.line(curvaCal.filter(d => d.serie === "curva nominal (K·x)"),
      { x: "x", y: "y", stroke: "serie", strokeDasharray: "6,4", strokeWidth: 1.5 }),
    Plot.line(curvaCal.filter(d => d.serie !== "curva nominal (K·x)"),
      { x: "x", y: "y", stroke: "serie", strokeWidth: 2.5 })
  ]
})
Tip

Con histéresis = 0 ambas ramas coinciden. La deriva de cero traslada la curva completa; la deriva de sensibilidad la “abre” proporcionalmente a la entrada, igual que en la Figura 1.3 de Webster.

Parámetros dinámicos

La mayoría de las variables biomédicas cambian en el tiempo, así que el instrumento se modela con ecuaciones diferenciales. Según el orden de la ecuación se habla de instrumentos de orden 0, 1, 2, …

Instrumento de primer orden

\[\tau \frac{dy(t)}{dt} + y(t) = K\,x(t), \qquad \tau = \frac{a_1}{a_0},\; K = \frac{b_0}{a_0}\]

Ejemplo: un filtro pasabajos RC (\(\tau = RC\)) o un termistor midiendo temperatura corporal (la constante de tiempo depende de su masa térmica).

viewof tau = Inputs.range([0.05, 3], { step: 0.05, value: 0.5, label: "Constante de tiempo τ (s)" })
paso1er = {
  const pts = [];
  const T = 6;
  for (let i = 0; i <= 400; i++) {
    const t = T * i / 400;
    pts.push({ t, y: 1 - Math.exp(-t / tau) });
  }
  return pts;
}

Plot.plot({
  style: estiloPlot,
  height: 280,
  grid: true,
  x: { label: "Tiempo (s)", domain: [0, 6] },
  y: { label: "y(t)/K (escalón unitario)", domain: [0, 1.15] },
  marks: [
    Plot.ruleY([1], { strokeDasharray: "4,4", strokeOpacity: 0.6 }),
    Plot.line(paso1er, { x: "t", y: "y", stroke: "#d62728", strokeWidth: 2.5 }),
    Plot.ruleX([tau], { strokeDasharray: "4,4", stroke: "#4c78a8" }),
    Plot.text([{ x: tau, t: "t = τ  (63,2 %)" }], { x: "x", y: () => 0.06, text: "t", fill: "#4c78a8", textAnchor: "start", dx: 5, fontWeight: "bold" })
  ],
  caption: "Respuesta al escalón: y(t) = K(1 − e^(−t/τ))"
})
bode1er = {
  const pts = [];
  for (let i = 0; i <= 300; i++) {
    const f = 10 ** (-2 + 4 * i / 300);
    const mag = 1 / Math.sqrt(1 + (2 * Math.PI * f * tau) ** 2);
    pts.push({ f, dB: 20 * Math.log10(mag) });
  }
  return pts;
}

Con τ pequeño el instrumento responde rápido y su ancho de banda es grande; con τ grande la respuesta se hace lenta y las componentes de alta frecuencia de la señal fisiológica se atenúan.

Instrumento de segundo orden: el sistema catéter–sensor

\[\frac{1}{\omega_n^2}\frac{d^2y}{dt^2} + \frac{2\zeta}{\omega_n}\frac{dy}{dt} + y = K\,x(t)\]

Queda caracterizado por su sensibilidad estática \(K\), su frecuencia natural \(f_n = \omega_n/2\pi\) y su factor de amortiguamiento \(\zeta\). Es el modelo clásico del sistema catéter–sensor para medir presión arterial. Los parámetros de inercia de la columna de líquido, la fricción y la elasticidad del catéter fijan \(f_n\) y \(\zeta\).

viewof fn = Inputs.range([2, 40], { step: 0.5, value: 15, label: "Frecuencia natural fₙ (Hz)" })
viewof zeta = Inputs.range([0.05, 2], { step: 0.05, value: 0.3, label: "Amortiguamiento ζ" })
paso2do = {
  const wn = 2 * Math.PI * fn;
  const T = Math.max(0.6, 8 / (Math.max(zeta, 0.08) * wn));
  const pts = [];
  for (let i = 0; i <= 600; i++) {
    const t = T * i / 600;
    let y;
    if (zeta < 0.999) {
      const wd = wn * Math.sqrt(1 - zeta * zeta);
      y = 1 - Math.exp(-zeta * wn * t) *
        (Math.cos(wd * t) + (zeta / Math.sqrt(1 - zeta * zeta)) * Math.sin(wd * t));
    } else if (zeta < 1.001) {
      y = 1 - Math.exp(-wn * t) * (1 + wn * t);
    } else {
      const s1 = -wn * (zeta - Math.sqrt(zeta * zeta - 1));
      const s2 = -wn * (zeta + Math.sqrt(zeta * zeta - 1));
      y = 1 - (s2 * Math.exp(s1 * t) - s1 * Math.exp(s2 * t)) / (s2 - s1);
    }
    pts.push({ t: 1000 * t, y });
  }
  return pts;
}

Plot.plot({
  style: estiloPlot,
  height: 280,
  grid: true,
  x: { label: "Tiempo (ms)" },
  y: { label: "y(t)/K (escalón unitario)", domain: [0, Math.max(1.15, 1.05 * Math.max(...paso2do.map(d => d.y)))] },
  marks: [
    Plot.ruleY([1], { strokeDasharray: "4,4", strokeOpacity: 0.6 }),
    Plot.line(paso2do, { x: "t", y: "y", stroke: "#d62728", strokeWidth: 2.5 })
  ],
  caption: "Respuesta al escalón (p. ej., prueba de «pop» o flush del catéter)"
})
bode2do = {
  const pts = [];
  for (let i = 0; i <= 300; i++) {
    const f = 10 ** (-1 + 3 * i / 300);
    const r = f / fn;
    const mag = 1 / Math.sqrt((1 - r * r) ** 2 + (2 * zeta * r) ** 2);
    pts.push({ f, dB: 20 * Math.log10(mag) });
  }
  return pts;
}

Plot.plot({
  style: estiloPlot,
  height: 280,
  grid: true,
  x: { label: "Frecuencia (Hz)", type: "log", domain: [0.1, 100] },
  y: { label: "|Y/X| /K (dB)", domain: [-40, 25] },
  marks: [
    Plot.ruleY([0], { strokeDasharray: "4,4", strokeOpacity: 0.6 }),
    Plot.line(bode2do, { x: "f", y: "dB", stroke: "#4c78a8", strokeWidth: 2.5 }),
    Plot.ruleX([fn], { strokeDasharray: "4,4", stroke: "#d62728" }),
    Plot.text([{ x: fn, t: `fₙ = ${fn} Hz` }], { x: "x", y: () => -35, text: "t", fill: "#d62728", textAnchor: "start", dx: 5, fontWeight: "bold" })
  ],
  caption: "Respuesta en frecuencia: con ζ pequeño aparece un pico de resonancia en fₙ"
})
  • ζ < 1 (subamortiguado): oscilaciones (“ringing”) y pico de resonancia.
  • ζ = 1 (crítico): la respuesta más rápida sin oscilar.
  • ζ > 1 (sobreamortiguado): respuesta lenta, atenúa las altas frecuencias.

Distorsión de la onda de presión arterial

Esta sección muestra la consecuencia clínica en la onda de presión arterial (75 lpm) medida a través de un sistema catéter–sensor con los \(f_n\) y \(\zeta\) elegidos arriba. La onda tiene armónicos significativos hasta ≈ 10–15 Hz; si \(f_n\) es baja o \(\zeta\) inadecuado, la onda medida se distorsiona. Un sistema subamortiguado sobreestima la sistólica (resonancia), uno sobreamortiguado la subestima y borra el nodo dicrótico.

armonicos = {
  // Onda arterial plantilla (un período, 75 lpm) descompuesta en serie de Fourier
  const N = 512, Kh = 18;
  const plantilla = (th) => {
    const dia = 78 + 14 * Math.exp(-2.2 * th);
    const sist = 37 * Math.exp(-(((th - 0.14) / 0.062) ** 2));
    const notch = -5 * Math.exp(-(((th - 0.34) / 0.022) ** 2));
    const dicrota = 8 * Math.exp(-(((th - 0.40) / 0.045) ** 2));
    return dia + sist + notch + dicrota;
  };
  const p = Array.from({ length: N }, (_, i) => plantilla(i / N));
  const c0 = p.reduce((s, v) => s + v, 0) / N;
  const arm = [];
  for (let k = 1; k <= Kh; k++) {
    let re = 0, im = 0;
    for (let i = 0; i < N; i++) {
      re += p[i] * Math.cos(2 * Math.PI * k * i / N);
      im -= p[i] * Math.sin(2 * Math.PI * k * i / N);
    }
    re *= 2 / N; im *= 2 / N;
    arm.push({ k, A: Math.hypot(re, im), phi: Math.atan2(im, re) });
  }
  return { c0, arm };
}

ondas = {
  const f0 = 75 / 60;              // 75 lpm
  const { c0, arm } = armonicos;
  const H = (f) => {
    const r = f / fn;
    const den = { re: 1 - r * r, im: 2 * zeta * r };
    const mag = 1 / Math.hypot(den.re, den.im);
    const ph = -Math.atan2(den.im, den.re);
    return { mag, ph };
  };
  const pts = [];
  for (let i = 0; i <= 700; i++) {
    const t = 2 / f0 * i / 700;    // dos latidos
    let pin = c0, pout = c0;
    for (const { k, A, phi } of arm) {
      const w = 2 * Math.PI * k * f0;
      pin += A * Math.cos(w * t + phi);
      const { mag, ph } = H(k * f0);
      pout += A * mag * Math.cos(w * t + phi + ph);
    }
    pts.push({ t, p: pin, serie: "presión real" });
    pts.push({ t, p: pout, serie: "presión medida" });
  }
  return pts;
}

Plot.plot({
  style: estiloPlot,
  height: 340,
  grid: true,
  x: { label: "Tiempo (s)" },
  y: { label: "Presión (mmHg)" },
  color: { legend: true, domain: ["presión real", "presión medida"], range: ["#7f7f7f", "#d62728"] },
  marks: [
    Plot.line(ondas.filter(d => d.serie === "presión real"), { x: "t", y: "p", stroke: "serie", strokeWidth: 2, strokeDasharray: "6,4" }),
    Plot.line(ondas.filter(d => d.serie === "presión medida"), { x: "t", y: "p", stroke: "serie", strokeWidth: 2.5 })
  ]
})
md`Con los valores actuales: fₙ = **${fn} Hz**, ζ = **${zeta}**.
Sistólica real ≈ **${Math.max(...ondas.filter(d => d.serie === "presión real").map(d => d.p)).toFixed(0)} mmHg** →
sistólica medida ≈ **${Math.max(...ondas.filter(d => d.serie === "presión medida").map(d => d.p)).toFixed(0)} mmHg**.`
NotaRegla práctica

Para reproducir fielmente la onda de presión arterial se recomienda \(f_n\) alta (≫ 10× la frecuencia cardíaca) y \(\zeta \approx 0{,}6\text{–}0{,}7\): es el compromiso que maximiza el ancho de banda plano sin resonancia. Prueba fₙ = 8 Hz con ζ = 0,2 (catéter largo con burbuja de aire) y compáralo con fₙ = 30 Hz, ζ = 0,65.
